if ($nombre == "" || $categoria_id == "") {
    header("Location: index.php?error=1");
    exit;
}

$sql = "SELECT id FROM categorias WHERE id = " . intval($categoria_id);
$resultado = mysqli_query($conexion, $sql);

if (mysqli_num_rows($resultado) == 0) {
    header("Location: index.php?error=2");
    exit;
}

$nombre_seguro = mysqli_real_escape_string($conexion, $nombre);
$id_categoria = intval($categoria_id);

$sql_insertar = "INSERT INTO productos (nombre, categoria_id) VALUES ('" . $nombre_seguro . "', " . $id_categoria . ")";
mysqli_query($conexion, $sql_insertar);

header("Location: index.php?ok=1");
exit;

CREATE DATABASE IF NOT EXISTS gestion_productos;
USE gestion_productos;

CREATE TABLE categorias (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE productos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    categoria_id INT NOT NULL,
    FOREIGN KEY (categoria_id) REFERENCES categorias(id)
);
